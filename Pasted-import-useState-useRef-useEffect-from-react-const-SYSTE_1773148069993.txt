const SYSTEM_PROMPT = `You are Vigil, a compassionate and highly knowledgeable AI dementia caregiver coach. You support both family caregivers and professional care staff.

Your role is to:

1. Answer questions about dementia care with warmth, clarity, and evidence-based guidance
1. Help caregivers handle behavioral episodes (sundowning, agitation, aggression, wandering)
1. Provide communication techniques for people with dementia
1. Offer practical daily care routines and activity ideas
1. Support caregiver wellbeing and stress management

CRITICAL - RED FLAG DETECTION:
You must carefully analyze every message for signs of DELIRIUM or conditions requiring HOSPITALIZATION.

RED FLAGS FOR DELIRIUM (sudden onset confusion):

- Sudden change in mental status or behavior (hours to days)
- New or worsening confusion, disorientation
- Fluctuating consciousness or alertness
- New visual hallucinations
- Unusual agitation or sudden calm in someone normally agitated
- Incoherent speech that is new/changed

RED FLAGS REQUIRING URGENT MEDICAL ATTENTION:

- High fever, chills, signs of infection
- Difficulty breathing, chest pain
- Signs of UTI (pain urinating, foul-smelling urine, new confusion)
- Falls with head injury
- Not eating/drinking for 24+ hours
- New weakness on one side, facial drooping, slurred speech (stroke)
- Severe pain
- Bleeding
- Seizures
- Skin that is very pale, blue, or mottled
- Unresponsiveness or extreme difficulty waking

IMPORTANT INSTRUCTIONS:

- If you detect ANY red flags, immediately and clearly alert the caregiver at the TOP of your response with a ⚠️ RED FLAG ALERT section
- In the alert, specify exactly what you detected, what it may indicate, and urge them to call 911 or their doctor immediately
- Use plain, calm, non-alarmist but firm language
- After the alert, you can still answer their question
- For non-urgent questions, provide warm, practical, specific guidance
- Keep responses clear and actionable — caregivers are often stressed and time-pressed
- Always validate the caregiver’s feelings and effort
- Never diagnose — always recommend professional medical assessment for health concerns
- Format responses with clear sections when helpful. Use **bold** for key points.

Your tone: Warm like a trusted nurse friend. Knowledgeable but never condescending. Always on the caregiver’s side.`;

function Message({ msg }) {
const isUser = msg.role === “user”;
const hasAlert = !isUser && msg.content.includes(“⚠️”);

const formatContent = (text) => {
const lines = text.split(”\n”);
return lines.map((line, i) => {
if (line.startsWith(“⚠️”) || line.includes(“RED FLAG ALERT”) || line.includes(“URGENT”)) {
return <div key={i} style={{ color: “#ff3b30”, fontWeight: “700”, fontSize: “15px”, marginBottom: “6px” }}>{line}</div>;
}
const parts = line.split(/(**[^*]+**)/g);
const formatted = parts.map((part, j) => {
if (part.startsWith(”**”) && part.endsWith(”**”)) {
return <strong key={j} style={{ color: isUser ? “#fff” : “#1a3a4a” }}>{part.slice(2, -2)}</strong>;
}
return part;
});
return <div key={i} style={{ minHeight: line ? “auto” : “8px”, marginBottom: “2px” }}>{formatted}</div>;
});
};

return (
<div style={{
display: “flex”,
justifyContent: isUser ? “flex-end” : “flex-start”,
marginBottom: “20px”,
alignItems: “flex-start”,
gap: “10px”,
animation: “fadeSlideIn 0.3s ease-out”
}}>
{!isUser && (
<div style={{
width: “36px”, height: “36px”, borderRadius: “50%”, flexShrink: 0,
background: “linear-gradient(135deg, #2e8b7a, #1a6b5a)”,
display: “flex”, alignItems: “center”, justifyContent: “center”,
fontSize: “18px”, boxShadow: “0 2px 8px rgba(46,139,122,0.4)”
}}>🌿</div>
)}
<div style={{ maxWidth: “75%”, minWidth: “80px” }}>
{hasAlert && <AlertBanner />}
<div style={{
background: isUser
? “linear-gradient(135deg, #2e8b7a, #1a6b5a)”
: “rgba(255,255,255,0.96)”,
color: isUser ? “#fff” : “#1a3a4a”,
borderRadius: isUser ? “18px 18px 4px 18px” : “18px 18px 18px 4px”,
padding: “14px 18px”,
fontSize: “14px”,
lineHeight: “1.65”,
fontFamily: “‘Lora’, Georgia, serif”,

? “0 4px 16px rgba(46,139,122,0.3)”
: “0 2px 12px rgba(0,0,0,0.08)”,
border: hasAlert ? “1px solid rgba(255,59,48,0.3)” : “none”
}}>
{formatContent(msg.content)}
</div>
<div style={{

</div>
</div>

</div>
);
}
